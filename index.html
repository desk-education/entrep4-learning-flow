This step-by-step guide explains how to configure Google Sheets spreadsheets and Google Apps Script web applications to automatically log donor records and synchronized student assessment scores with a 5-minute anti-spam cooldown protection.

A GCash Donation Integration Script (ExamShuffle Pro Engine)

Open your Google Sheet (ID: 1iXfkwDMLDbVty8ph7nyIJ3CTinTWKDhH67XDLoe0Few) and paste this script in Extensions → Apps Script:

/**
 * ExamShuffle Pro - Donor Information Google Sheets Integration
 */
const SPREADSHEET_ID = '1iXfkwDMLDbVty8ph7nyIJ3CTinTWKDhH67XDLoe0Few';
const SHEET_NAME = 'Donor Records';
const HEADERS = ['Timestamp', 'Donor Name', 'Donor Type', 'Organization / Group', 'Country', 'Amount', 'Source'];

function doPost(e) {
  let lock;
  try {
    const params = (e && e.parameter) ? e.parameter : {};
    const donorName = clean_(params.donorName || params.name, 160);
    const donorType = clean_(params.donorType, 20);
    const organization = clean_(params.donorOrganization || params.organization, 200);
    const country = clean_(params.country, 100);
    const amount = clean_(params.amount, 60);
    const source = clean_(params.source || 'ExamShuffle Pro Support My Work', 120);

    if (!donorName || !country || !['Individual', 'Group'].includes(donorType)) {
      return jsonResponse_({ success: false, message: 'Required donor fields missing.' });
    }

    lock = LockService.getScriptLock();
    lock.waitLock(10000);

    const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = spreadsheet.getSheetByName(SHEET_NAME);
    if (!sheet) { sheet = spreadsheet.insertSheet(SHEET_NAME); }

    if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
      sheet.setFrozenRows(1);
    }

    sheet.appendRow([
      new Date(),
      safeCell_(donorName),
      donorType,
      safeCell_(donorType === 'Group' ? organization : ''),
      safeCell_(country),
      safeCell_(amount),
      safeCell_(source)
    ]);

    SpreadsheetApp.flush();
    return jsonResponse_({ success: true, message: 'Donor record saved.', sheet: SHEET_NAME, row: sheet.getLastRow() });
  } catch (error) {
    return jsonResponse_({ success: false, message: String(error && error.message ? error.message : 'Error') });
  } finally {
    if (lock) { try { lock.releaseLock(); } catch(e){} }
  }
}

function clean_(val, maxLen) { return String(val || '').replace(/[\u0000-\u001F\u007F]/g, '').trim().slice(0, maxLen); }
function safeCell_(val) { const text = String(val || ''); return /^[=+\-@]/.test(text) ? "'" + text : text; }
function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }

B Assessment Quiz Synchronization Script (With 5-Min Cooldown Rate-Limiter)

Create another script project or sheet tab titled Quiz Scores and deploy the following code. This code enforces a mandatory 5-minute submission gap per student to prevent spam submissions:

/**
 * Student Assessment Quiz Score Integrator with 5-Minute Rate Limiting
 */
const SPREADSHEET_ID = '1iXfkwDMLDbVty8ph7nyIJ3CTinTWKDhH67XDLoe0Few';
const SHEET_NAME = 'Quiz Scores';
const HEADERS = ['Timestamp', 'Student Name', 'Grade & Section', 'Score', 'Total Questions', 'Percentage', 'Status'];
const COOLDOWN_MINUTES = 5;

function doPost(e) {
  let lock;
  try {
    const params = (e && e.parameter) ? e.parameter : {};
    const studentName = clean_(params.studentName, 150);
    const section = clean_(params.section, 100);
    const score = clean_(params.score, 10);
    const total = clean_(params.totalQuestions, 10);
    const percentage = clean_(params.percentage, 20);

    if (!studentName || !section) {
      return jsonResponse_({ success: false, message: 'Student Name and Section are required.' });
    }

    // Server-side Rate Limiting check via CacheService (5 minutes = 300 seconds)
    const cache = CacheService.getScriptCache();
    const cacheKey = 'quiz_sub_' + encodeURIComponent((studentName + '_' + section).toLowerCase().replace(/\s+/g, ''));
    const lastSubTime = cache.get(cacheKey);

    if (lastSubTime) {
      const remainingSec = Math.ceil((parseInt(lastSubTime) + (COOLDOWN_MINUTES * 60 * 1000) - Date.now()) / 1000);
      if (remainingSec > 0) {
        return jsonResponse_({ 
          success: false, 
          rateLimited: true,
          message: 'Multiple submissions restricted. Please wait ' + Math.ceil(remainingSec / 60) + ' more minute(s) before submitting again.' 
        });
      }
    }

    lock = LockService.getScriptLock();
    lock.waitLock(10000);

    const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = spreadsheet.getSheetByName(SHEET_NAME);
    if (!sheet) { sheet = spreadsheet.insertSheet(SHEET_NAME); }

    if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
      sheet.setFrozenRows(1);
    }

    const numScore = parseFloat(score);
    const statusStr = numScore >= 10 ? "PASSED" : "NEEDS IMPROVEMENT";

    sheet.appendRow([
      new Date(),
      safeCell_(studentName),
      safeCell_(section),
      numScore,
      parseFloat(total),
      percentage + "%",
      statusStr
    ]);

    // Store timestamp in cache for 5 minutes (300 seconds)
    cache.put(cacheKey, Date.now().toString(), 300);

    SpreadsheetApp.flush();
    return jsonResponse_({ success: true, message: 'Quiz score synchronized successfully!' });
  } catch (error) {
    return jsonResponse_({ success: false, message: String(error.message || 'Error saving quiz score') });
  } finally {
    if (lock) { try { lock.releaseLock(); } catch(e){} }
  }
}

function clean_(val, maxLen) { return String(val || '').replace(/[\u0000-\u001F\u007F]/g, '').trim().slice(0, maxLen); }
function safeCell_(val) { const text = String(val || ''); return /^[=+\-@]/.test(text) ? "'" + text : text; }
function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
